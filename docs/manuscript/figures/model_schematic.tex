% =====================================================================
% Electrical analogue of the 14-state closed-loop lumped-parameter model.
% Topology transcribed from src/models/system_rhs.m (states and equations),
% src/models/valve_model.m and src/models/vsd_shunt_model.m.
% Blue elements are the seven parameters selected by the global sensitivity
% analysis (S_T >= 0.10) and fitted during calibration; grey elements are
% held at their allometrically scaled values.
%
% Compile:  pdflatex model_schematic.tex
% =====================================================================
\documentclass[border=8pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{amsmath}
\usepackage[american]{circuitikz}
\usetikzlibrary{arrows.meta, fit, backgrounds, positioning}

\definecolor{calib}{RGB}{0,84,159}
\definecolor{shunt}{RGB}{190,30,45}
\definecolor{fixedel}{RGB}{70,70,70}

\ctikzset{bipoles/length=1.0cm}
\ctikzset{resistors/scale=0.85, capacitors/scale=0.95, inductors/scale=0.8, diodes/scale=0.8}
\ctikzset{label/align = straight}
\tikzset{chamber/.style={font=\small\bfseries}}

\begin{document}
\begin{circuitikz}[line width=0.7pt, font=\footnotesize, color=fixedel]

% ---------------------------------------------------------------- heart
\coordinate (LA) at (0,0);
\coordinate (LV) at (3,0);
\coordinate (RV) at (7,0);
\coordinate (RA) at (10,0);

\draw (LA) to[D*, l=MV, *-*] (LV);
\draw (RA) to[D*, l_=TV, *-*] (RV);

\draw (LA) to[vC, l_=$E_{LA}(t)$, a^=$P_{LA}$] (0,-2.1) node[ground]{};
\draw (LV) to[vC, l_=$E_{LV}(t)$, a^=$P_{LV}$] (3,-2.1) node[ground]{};
\draw[calib] (RV) to[vC, l=$E_{RV}(t)$, a_=$P_{RV}$] (7,2.1) node[ground, rotate=180]{};
\draw (RA) to[vC, l=$E_{RA}(t)$, a_=$P_{RA}$] (10,2.1) node[ground, rotate=180]{};

\node[chamber, above left=2pt and 1pt of LA] {LA};
\node[chamber, above left=2pt and 1pt of LV] {LV};
\node[chamber, below right=2pt and 1pt of RV] {RV};
\node[chamber, below right=2pt and 1pt of RA] {RA};
\node[calib, font=\scriptsize, anchor=south] at (7.0,2.8) {$E_{B,RV}$ fitted};

% ------------------------------------------------------------- VSD shunt
\draw[shunt, line width=0.9pt] (LV) to[vR, l=$C_d A_{\mathrm{VSD}}$, a_=$Q_{\mathrm{VSD}}$] (RV);
\node[shunt, font=\scriptsize] at (5,-1.15) {VSD orifice};

% ------------------------------------------------ systemic circulation
\draw (LV) -- (3,2.2);
\draw (3,2.2) to[D*, l_=AV] (1.2,2.2) coordinate (SAR);
\draw[calib] (SAR) to[C, l=$C_{SAR}$, a_=$P_{SAR}$] (1.2,4.0) node[ground, rotate=180]{};
\draw[calib] (SAR) to[R, l_=$R_{SAR}$] (-0.6,2.2);
\draw (-0.6,2.2) to[L, l_=$L_{SAR}$, a^=$Q_{SAR}$] (-2.4,2.2);
\draw[calib] (-2.4,2.2) to[R, l_=$R_{SC}/2$] (-4.2,2.2) coordinate (SC);
\draw (SC) to[C, l=$C_{SC}$, a_=$P_{SC}$] (-4.2,4.0) node[ground, rotate=180]{};
\draw[calib] (SC) to[R, l_=$R_{SC}/2$] (-6.0,2.2);
\draw (-6.0,2.2) -- (-6.0,6.0) -- (-4.8,6.0) coordinate (SVEN);
\draw (SVEN) to[C, l=$C_{SVEN}$, a_=$P_{SVEN}$] (-4.8,7.8) node[ground, rotate=180]{};
\draw[calib] (SVEN) to[R, l_=$R_{SVEN}$] (-3.0,6.0);
\draw (-3.0,6.0) to[L, l_=$L_{SVEN}$, a^=$Q_{SVEN}$] (-1.2,6.0);
\draw (-1.2,6.0) -- (11,6.0) -- (11,0) -- (RA);

\node[font=\small\itshape, text=black!70] at (2.0,7.2) {systemic circulation};
\draw[-{Stealth[length=5pt]}] (4.2,6.0) -- (5.0,6.0);
\draw[-{Stealth[length=5pt]}] (3,0.9) -- (3,1.7);

% ----------------------------------------------- pulmonary circulation
\draw (RV) -- (7,-2.4);
\draw (7,-2.4) to[D*, l=PV] (8.8,-2.4) coordinate (PAR);
\draw[calib] (PAR) to[C, l_=$C_{PAR}$, a^=$P_{PAR}$] (8.8,-4.2) node[ground]{};
\draw[calib] (PAR) to[R, l=$R_{PAR}$] (10.6,-2.4);
\draw (10.6,-2.4) to[L, l=$L_{PAR}$, a_=$Q_{PAR}$] (12.4,-2.4) coordinate (PC);
\draw (PC) to[C, l_=$C_{PC}$, a^=$P_{PC}$] (12.4,-4.2) node[ground]{};
\draw (PC) -- (13.8,-2.4);
\draw[calib] (13.8,-2.4) -- (13.8,-1.4) to[R, l=$R_{PC,OX}$] (16.0,-1.4) -- (16.0,-2.4);
\draw (13.8,-2.4) -- (13.8,-3.4) to[R, l_=$R_{PC,NO}$] (16.0,-3.4) -- (16.0,-2.4);
\draw (16.0,-2.4) -- (17.6,-2.4) -- (17.6,-6.2) -- (16.4,-6.2) coordinate (PVEN);
\draw (PVEN) to[C, l_=$C_{PVEN}$, a^=$P_{PVEN}$] (16.4,-8.0) node[ground]{};
\draw[calib] (PVEN) to[R, l=$R_{PVEN}$] (14.6,-6.2);
\draw (14.6,-6.2) to[L, l=$L_{PVEN}$, a_=$Q_{PVEN}$] (12.8,-6.2);
\draw (12.8,-6.2) -- (-1.2,-6.2) -- (-1.2,0) -- (LA);

\node[font=\small\itshape, text=black!70] at (11.0,-7.6) {pulmonary circulation};
\draw[-{Stealth[length=5pt]}] (5.0,-6.2) -- (4.2,-6.2);
\draw[-{Stealth[length=5pt]}] (7,-1.2) -- (7,-2.0);

% ---------------------------------------------------------------- legend
\draw[fixedel] (-6.0,-5.0) -- (-5.2,-5.0);
\node[anchor=west, font=\scriptsize] at (-5.1,-5.0) {fixed by allometric scaling};
\draw[calib, line width=0.9pt] (-6.0,-5.7) -- (-5.2,-5.7);
\node[anchor=west, font=\scriptsize] at (-5.1,-5.7) {calibrated ($S_T \ge 0.10$)};
\draw[shunt, line width=0.9pt] (-6.0,-6.4) -- (-5.2,-6.4);
\node[anchor=west, font=\scriptsize] at (-5.1,-6.4) {ventricular septal defect ($C_d$ calibrated)};

\end{circuitikz}
\end{document}
